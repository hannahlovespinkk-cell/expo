import React, { useEffect, useState } from "react";
import {
  SafeAreaView, View, Text, Pressable, TextInput, ScrollView,
  Switch, Alert, StyleSheet, StatusBar
} from "react-native";
import AsyncStorage from "@react-native-async-storage/async-storage";

const STORAGE_KEY = "afterhours_guard_v1";
const ACTIONS = [
  { id: "ex", title: "Texting an ex", detail: "Put the message in a draft instead." },
  { id: "posts", title: "Posting or going live", detail: "Save it for tomorrow-you to review." },
  { id: "dm", title: "Sending risky DMs", detail: "Pause before sending anything personal." },
  { id: "shopping", title: "Impulse shopping", detail: "Keep your card details out of reach." },
  { id: "calls", title: "Calling someone impulsively", detail: "Write down what you want to say." },
  { id: "screenshots", title: "Sharing private screenshots", detail: "Never share private or intimate content without consent." },
];

const DURATIONS = [
  { label: "1 hour", hours: 1 },
  { label: "3 hours", hours: 3 },
  { label: "6 hours", hours: 6 },
  { label: "Until morning", hours: 10 },
];

function remaining(ms) {
  if (ms <= 0) return "00:00";
  const total = Math.ceil(ms / 1000);
  const h = Math.floor(total / 3600);
  const m = Math.floor((total % 3600) / 60);
  const s = total % 60;
  return h > 0
    ? `${h}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`
    : `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
}

export default function App() {
  const [modeUntil, setModeUntil] = useState(0);
  const [now, setNow] = useState(Date.now());
  const [enabled, setEnabled] = useState({ ex: true, posts: true, dm: true, shopping: true, calls: true, screenshots: true });
  const [contact, setContact] = useState("");
  const [draft, setDraft] = useState("");
  const [drafts, setDrafts] = useState([]);
  const [urge, setUrge] = useState(false);
  const [urgeUntil, setUrgeUntil] = useState(0);
  const [setupDone, setSetupDone] = useState(false);

  useEffect(() => {
    (async () => {
      try {
        const raw = await AsyncStorage.getItem(STORAGE_KEY);
        if (raw) {
          const data = JSON.parse(raw);
          setModeUntil(data.modeUntil || 0);
          setEnabled(data.enabled || enabled);
          setContact(data.contact || "");
          setDrafts(data.drafts || []);
          setSetupDone(true);
        }
      } catch (_) {}
    })();
  }, []);

  useEffect(() => {
    const timer = setInterval(() => setNow(Date.now()), 1000);
    return () => clearInterval(timer);
  }, []);

  useEffect(() => {
    AsyncStorage.setItem(STORAGE_KEY, JSON.stringify({ modeUntil, enabled, contact, drafts })).catch(() => {});
  }, [modeUntil, enabled, contact, drafts]);

  useEffect(() => {
    if (urge && now >= urgeUntil) setUrge(false);
  }, [now, urge, urgeUntil]);

  const active = modeUntil > now;

  function startMode(hours) {
    setModeUntil(Date.now() + hours * 60 * 60 * 1000);
    setSetupDone(true);
    setUrge(false);
    Alert.alert("Guard mode is on ð¡ï¸", "Your pause timer has started. This app helps you slow down, save drafts, and remember your boundaries.");
  }

  function endMode() {
    Alert.alert(
      "End Guard mode?",
      "Future-you asked for this protection. If youâre feeling impaired, the safest choice is to keep the timer running.",
      [
        { text: "Keep it on", style: "cancel" },
        { text: "End anyway", style: "destructive", onPress: () => setModeUntil(0) }
      ]
    );
  }

  function saveDraft() {
    if (!draft.trim()) {
      Alert.alert("Nothing to save", "Write what you want to say first.");
      return;
    }
    setDrafts([{ id: String(Date.now()), body: draft.trim(), createdAt: new Date().toLocaleString() }, ...drafts]);
    setDraft("");
    Alert.alert("Saved for tomorrow ð", "Your message is stored here, not sent to anyone. Re-read it when youâre clear-headed.");
  }

  function deleteDraft(id) {
    Alert.alert("Delete this draft?", "This cannot be undone.", [
      { text: "Keep", style: "cancel" },
      { text: "Delete", style: "destructive", onPress: () => setDrafts(drafts.filter(d => d.id !== id)) }
    ]);
  }

  function startUrgePause() {
    setUrge(true);
    setUrgeUntil(Date.now() + 20 * 1000);
  }

  return (
    <SafeAreaView style={styles.safe}>
      <StatusBar barStyle="light-content" />
      <ScrollView contentContainerStyle={styles.container}>
        <View style={styles.hero}>
          <Text style={styles.eyebrow}>YOUR FUTURE SELF SAYS THANK YOU</Text>
          <Text style={styles.logo}>afterhours<Text style={styles.logoDot}>.</Text></Text>
          <Text style={styles.subtitle}>A little space between the impulse and the send button.</Text>
          <View style={[styles.status, active ? styles.statusOn : styles.statusOff]}>
            <View style={[styles.statusDot, active ? styles.dotOn : styles.dotOff]} />
            <Text style={styles.statusText}>{active ? `GUARD MODE ON Â· ${remaining(modeUntil - now)} LEFT` : "GUARD MODE OFF"}</Text>
          </View>
        </View>

        <View style={styles.card}>
          <Text style={styles.sectionTitle}>ð¡ï¸ Set your guard</Text>
          <Text style={styles.muted}>Choose how long you want your future self to be protected.</Text>
          <View style={styles.durationGrid}>
            {DURATIONS.map(item => (
              <Pressable key={item.label} onPress={() => startMode(item.hours)} style={({ pressed }) => [styles.duration, pressed && styles.pressed, active && modeUntil - now > 0 && Math.abs((modeUntil - now) - item.hours * 3600000) < 60000 && styles.durationSelected]}>
                <Text style={styles.durationLabel}>{item.label}</Text>
                <Text style={styles.durationSub}>{item.hours === 10 ? "Sleep on it" : "Stay grounded"}</Text>
              </Pressable>
            ))}
          </View>
          {active && <Pressable onPress={endMode} style={styles.textButton}><Text style={styles.textButtonText}>End guard mode early</Text></Pressable>}
        </View>

        <View style={styles.card}>
          <Text style={styles.sectionTitle}>ð« Your no-regrets list</Text>
          <Text style={styles.muted}>Choose the situations you want reminders for while Guard mode is on.</Text>
          {ACTIONS.map((item, index) => (
            <View key={item.id} style={[styles.actionRow, index === ACTIONS.length - 1 && { borderBottomWidth: 0 }]}>
              <View style={styles.actionCopy}>
                <Text style={styles.actionTitle}>{item.title}</Text>
                <Text style={styles.actionDetail}>{item.detail}</Text>
              </View>
              <Switch value={!!enabled[item.id]} onValueChange={v => setEnabled({ ...enabled, [item.id]: v })} trackColor={{ false: "#34343d", true: "#bca6ff" }} thumbColor={enabled[item.id] ? "#ffffff" : "#9999a5"} />
            </View>
          ))}
          <Text style={styles.smallNote}>These are personal reminders. Phone operating systems do not let an ordinary app silently block every other app or intercept messages from Instagram, iMessage, TikTok, etc.</Text>
        </View>

        <View style={styles.card}>
          <Text style={styles.sectionTitle}>ð The unsent-message vault</Text>
          <Text style={styles.muted}>Say it here. Don't send it tonight.</Text>
          <TextInput
            style={styles.input}
            placeholder="Type the message you really want to sendâ¦"
            placeholderTextColor="#858392"
            value={draft}
            onChangeText={setDraft}
            multiline
            textAlignVertical="top"
          />
          <Pressable style={styles.primaryButton} onPress={saveDraft}>
            <Text style={styles.primaryButtonText}>Save it for tomorrow</Text>
          </Pressable>
          {drafts.length > 0 && <Text style={styles.subhead}>Saved drafts ({drafts.length})</Text>}
          {drafts.map(item => (
            <View key={item.id} style={styles.draftCard}>
              <Text style={styles.draftDate}>{item.createdAt}</Text>
              <Text style={styles.draftBody}>{item.body}</Text>
              <Pressable onPress={() => deleteDraft(item.id)}><Text style={styles.deleteText}>Delete draft</Text></Pressable>
            </View>
          ))}
        </View>

        <View style={styles.card}>
          <Text style={styles.sectionTitle}>â¸ï¸ The 20-second reset</Text>
          <Text style={styles.muted}>For when youâre about to do something you might regret.</Text>
          {!urge ? (
            <Pressable style={styles.secondaryButton} onPress={startUrgePause}>
              <Text style={styles.secondaryButtonText}>I have an urge â pause me</Text>
            </Pressable>
          ) : (
            <View style={styles.pauseBox}>
              <Text style={styles.pauseNumber}>{Math.max(0, Math.ceil((urgeUntil - now) / 1000))}</Text>
              <Text style={styles.pauseHeadline}>Breathe. Nothing needs to happen right now.</Text>
              <Text style={styles.pauseText}>Put your phone face down. Take a slow breath in, then a longer breath out. Ask: âWill I be glad I did this tomorrow?â</Text>
              <Pressable onPress={() => setUrge(false)} style={styles.textButton}><Text style={styles.textButtonText}>I feel calmer</Text></Pressable>
            </View>
          )}
        </View>

        <View style={styles.card}>
          <Text style={styles.sectionTitle}>ð¤ Your real-life safety plan</Text>
          <Text style={styles.muted}>A trusted person can help you get home safely or hold onto your phone for a while.</Text>
          <TextInput
            style={styles.singleInput}
            placeholder="Trusted person's name or number (optional)"
            placeholderTextColor="#858392"
            value={contact}
            onChangeText={setContact}
            autoCapitalize="words"
          />
          <Text style={styles.smallNote}>This saves a reminder only; it does not contact them automatically. If youâve been drinking, donât drive. Stay with someone you trust and arrange a sober ride.</Text>
        </View>

        <View style={styles.footer}>
          <Text style={styles.footerTitle}>A reminder from tomorrow-you</Text>
          <Text style={styles.footerQuote}>âYou don't have to act on every feeling. You can feel it, write it down, and decide later.â</Text>
          <Text style={styles.footerTiny}>AFTERHOURS Â· YOUR PEACE OVER YOUR IMPULSE</Text>
        </View>
      </ScrollView>
    </SafeAreaView>
  );
}
